PaymentProcessingService.java 
@RestController
@RequestMapping("/api/v1/payments")
@RequiredArgsConstructor
public class PaymentController {

    private final PaymentService paymentService;
    private final KafkaTemplate<String, PaymentEvent> kafka;

    @PostMapping("/process")
    @Transactional(isolation = Isolation.SERIALIZABLE)
    public ResponseEntity<PaymentResponse> execute(
            @Valid @RequestBody PaymentRequest req) {
        
        // Idempotent Transaction Execution
        Payment tx = paymentService.process(req);
        kafka.send("payment-events", tx.toEvent());
        
        return ResponseEntity.ok(new PaymentResponse(tx));
    }
}
